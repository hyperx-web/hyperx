% ================= rows =================
\def\yT{0}       % transmitter chain
\def\yR{-2.35}   % receiver chain
\def\yS{-3.55}   % sensing branch
\def\xL{-0.6}
\def\xR{10.15}

% ---------- background panels ----------
\begin{scope}[on background layer]
  \draw[grpbox, fill=txfill] (\xL,0.95) rectangle (\xR,-0.75);
  \draw[grpbox, fill=rxfill] (\xL,-1.45) rectangle (\xR,-4.25);
\end{scope}
\node[grplab, anchor=north west] at (\xL+0.05,0.92) {Transmitter};
\node[grplab, anchor=north west] at (\xL+0.05,-1.48) {Receiver};

% ================= transmitter =================
\node[lab, anchor=east] (bits) at (0.15,\yT) {$\mathbf b$};
\node[blk, minimum width=12mm] (qam) at (1.05,\yT) {QAM\\mapping};
\node[cblk, minimum width=21mm] (idaft) at (3.55,\yT) {IDAFT\\[-1pt]$\mathbf A^{\mathsf H}\!=\!\bm\Lambda_{c_1}^{\mathsf H}\mathbf F^{\mathsf H}\bm\Lambda_{c_2}^{\mathsf H}$};
\node[cblk, minimum width=14mm] (cpp) at (6.05,\yT) {Add\\CPP};
\node[blk, minimum width=12mm] (dac) at (7.75,\yT) {P/S\\DAC};

\draw[line,->] (bits) -- (qam);
\draw[cline,->] (qam) -- node[lab, above] {$\mathbf x$} (idaft);
\draw[cline,->] (idaft) -- node[lab, above] {$\mathbf s$} (cpp);
\draw[line,->] (cpp) -- (dac);

% tx antenna
\coordinate (txa) at (9.05,\yT);
\draw[line] (dac.east) -- (txa);
\draw[line] (txa) -- ++(0,0.32) -- ++(-0.16,0.22) ++(0.16,-0.22) -- ++(0.16,0.22)
  ($(txa)+(0,0.32)$) circle (0.5pt);   % one path = antenna symbol

% chirp parameters
\node[sm, text=cedge, below=1.5pt of idaft] {chirp parameters $c_1, c_2$};

% ================= channel =================
\coordinate (rxa) at (9.05,\yR);
\coordinate (ch1) at (9.75,{\yT+0.54});
\coordinate (ch2) at (9.75,{\yR+0.54});
\draw[line] ($(txa)+(0.2,0.54)$) -- (ch1);
\draw[line] (ch1) -- (ch2);
\draw[line,->] (ch2) -- ($(rxa)+(0.2,0.54)$);
\node[lab, anchor=west, align=left] at (10.2,{(\yT+\yR)/2+0.54})
  {LTV channel\\[1pt]\footnotesize $\{h_i,\ell_i,\nu_i\}_{i=1}^{P}$};

% ================= receiver =================
\coordinate (rxa) at (9.05,\yR);
\draw[line] (rxa) -- ++(0,0.32) -- ++(-0.16,0.22) ++(0.16,-0.22) -- ++(0.16,0.22)
  ($(rxa)+(0,0.32)$) circle (0.5pt);   % one path = antenna symbol

\node[blk, minimum width=12mm] (adc) at (7.75,\yR) {ADC\\S/P};
\node[cblk, minimum width=14mm] (rcpp) at (6.05,\yR) {Remove\\CPP};
\node[cblk, minimum width=21mm] (daft) at (3.55,\yR) {DAFT\\[-1pt]$\mathbf A\!=\!\bm\Lambda_{c_2}\mathbf F\bm\Lambda_{c_1}$};
\node[blk, minimum width=12mm] (det) at (1.05,\yR) {Detection\\demapping};
\node[lab, anchor=east] (bhat) at (-0.05,\yR) {$\widehat{\mathbf b}$};

\draw[line,->] (rxa) -- (adc);
\draw[line,->] (adc) -- (rcpp);
\draw[cline,->] (rcpp) -- node[lab, above] {$\mathbf r$} (daft);
\coordinate (yj) at (2.05,\yR);
\draw[cline] (daft) -- (yj);
\draw[cline,->] (yj) -- (det);
\node[lab, above] at ($(daft.west)!0.5!(yj)$) {$\mathbf y$};
\draw[line,->] (det) -- (bhat);
\node[dot] at (yj) {};

% ================= sensing branch (ISAC) =================
\node[sblk, minimum width=30mm] (sens) at (5.0,\yS) {Delay--Doppler estimation};
\coordinate (sj) at (2.05,\yS);
\draw[sline] (yj) -- (sj);
\draw[sline,->] (sj) -- (sens);
\node[lab, anchor=west, text=scol] (est) at (7.15,\yS) {$\{\hat\ell_i,\hat\nu_i\}$};
\draw[sline,->] (sens) -- (est);
\node[sm, text=scol, above] at ($(sj)!0.5!(sens.west)$) {sensing};

% effective channel annotation
\node[sm, below=1.5pt of daft] {$\mathbf y=\mathbf H_{\mathrm{eff}}\mathbf x+\mathbf w$};
